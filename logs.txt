# 1. Active space
quenais-run --molecule ScF --basis sto-3g --steps 0 1 --project-dir runs/ScF_las \
  --active-space-method avas --avas-ao-labels 'Sc 3d' 'Sc 4s' 'F 2p'

# 2. Classical reference (LASSCF)
quenais-run --molecule ScF --basis sto-3g --steps 3 --project-dir runs/ScF_las \
  --active-space-method avas --avas-ao-labels 'Sc 3d' 'Sc 4s' 'F 2p' \
  --solver lasscf --las-fragments 'Sc:6:1,1' 'F:3:3,3'

# 3. Quantum version (LASSQD) + plots — quick setting first
quenais-run --molecule ScF --basis sto-3g --steps 3 4 --project-dir runs/ScF_las \
  --active-space-method avas --avas-ao-labels 'Sc 3d' 'Sc 4s' 'F 2p' \
  --solver lassqd --las-fragments 'Sc:6:1,1' 'F:3:3,3' --lassqd-shots 20000
