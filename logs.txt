import csv, glob, os, pickle
B = os.environ["B"]
seen = set()
for grid in ("h_chain", "n2_curve", "blind"):
    rows = {r["run_id"]: r for r in csv.DictReader(open(f"{B}/{grid}/results.csv"))}
    for rid, r in rows.items():
        key = (r["molecule"], r["tag_R"], r["basis"])
        f = f"{B}/{grid}/runs/{rid}/results/step1_asf.pkl"
        if key in seen or not os.path.exists(f):
            continue
        seen.add(key)
        d = pickle.load(open(f, "rb"))
        e_uhf = d.get("uhf_energy")
        e_rhf = float(r["E_HF_Ha"]) if r.get("E_HF_Ha") else None
        diff = (e_uhf - e_rhf) * 627.5 if (e_uhf is not None and e_rhf) else None
        print(f"{key[0]:8s} {key[1]:>5s} {key[2]:8s}  UHF {e_uhf}  RHF {e_rhf}  UHF-RHF {diff} kcal/mol")
